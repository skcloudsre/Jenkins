aws ec2 describe-security-groups --group-id sg-0fbd9214b95bdae92 --query 'SecurityGroups[0].IpPermissions[*].[IpProtocol,FromPort,ToPort,IpRanges[*].CidrIp]' --output table --region ap-south-1
#See only inbound rules easier-to-read output

aws ec2 authorize-security-group-ingress \
  --group-id sg-0123456789abcdef \
  --ip-permissions \
  IpProtocol=tcp,FromPort=22,ToPort=22,IpRanges='[{CidrIp=0.0.0.0/0,Description="SSH"}]' \
  IpProtocol=tcp,FromPort=80,ToPort=80,IpRanges='[{CidrIp=0.0.0.0/0,Description="HTTP"}]' \
  IpProtocol=-1,IpRanges='[{CidrIp=0.0.0.0/0,Description="All Traffic"}]' 
or
aws ec2 authorize-security-group-ingress --group-id sg-0fbd9214b95bdae92 --ip-permissions IpProtocol=tcp,FromPort=22,ToPort=22,IpRanges='[{CidrIp=0.0.0.0/0,Description="SSH"}]' IpProtocol=tcp,FromPort=80,ToPort=80,IpRanges='[{CidrIp=0.0.0.0/0,Description="HTTP"}]' IpProtocol=-1,IpRanges='[{CidrIp=0.0.0.0/0,Description="All Traffic"}]' --region ap-south-1
#add SSH (22), HTTP (80), and All Traffic inbound rules in a single AWS CLI command using authorize-security-group-ingress
