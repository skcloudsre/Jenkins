aws ec2 create-key-pair `
  --key-name test-kp `
  --key-type rsa `
  --key-format pem `
  --region ap-south-1 `
  --query "KeyMaterial" `
  --output text | Out-File -Encoding ascii "C:\Users\SESHU_VANI\Downloads\test-kp.pem"
#create Key Pair with Rsa Algorithm

aws ec2 describe-key-pairs `
  --key-names test-kp `
  --region ap-south-1 `
  --query "KeyPairs[0]" `
  --output json
#Verify the key pair 

aws ec2 describe-key-pairs --key-names test-kp --region ap-south-1 --query "KeyPairs[0].[KeyName,KeyPairId,KeyType]" --output table
